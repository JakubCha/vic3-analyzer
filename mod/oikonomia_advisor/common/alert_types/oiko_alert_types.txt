# Oikonomia Advisor alerts in the top bar (format: game/common/alert_types/00_alert_types.txt).

# The state's construction queue runs dry: construction sectors are paid for and idle.
oiko_idle_construction_alert = {
	texture = "gfx/interface/icons/alert_icons/wasted_construction.dds"
	script_context = player_country

	valid = {
		oiko_construction_levels >= 1
		construction_queue_government_duration < 2
		in_default = no
	}

	type = important_action
	color = yellow
	open_panel = construction|default
}

# Debt past half the credit limit.
oiko_debt_half_alert = {
	texture = "gfx/interface/icons/alert_icons/bankruptcy.dds"
	script_context = player_country

	valid = {
		scaled_debt >= 0.5
		in_default = no
	}

	type = angry_important_action
	color = red
	open_panel = budget|default
}

# The deficit pays the army, not growth.
oiko_upkeep_deficit_alert = {
	texture = "gfx/interface/icons/event_icons/event_scales.dds"
	script_context = player_country

	valid = {
		net_fixed_income < 0
		military_expenses_share >= 0.3
	}

	type = important_action
	color = yellow
	open_panel = budget|default
}
