# Oikonomia Advisor: values shown in the journal entries. Root = the country.
# Flags are 1/0 so localization can tick checkboxes by comparing to oiko_one.

oiko_zero = { value = 0 }
oiko_one = { value = 1 }

# --- Budget guard ---

oiko_balance = { value = net_fixed_income }
oiko_treasury = { value = gold_reserves }

# Weeks until the treasury is empty at the current deficit (0 when not losing money).
oiko_weeks_to_debt = {
	value = 0
	if = {
		limit = {
			net_fixed_income < 0
			gold_reserves > 0
		}
		value = gold_reserves
		divide = net_fixed_income
		multiply = -1
	}
}

# Principal relative to the credit limit (1 = default).
oiko_debt_share = { value = scaled_debt }

oiko_military_share = { value = military_expenses_share }
oiko_military_target = { value = 0.4 }

	}

oiko_construction_levels = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_construction_sector }
		add = this.level
	}
}
oiko_construction_target = { value = 10 }

# --- Live link (scripted_effects/oiko_effects.txt) ---

oiko_income = { value = fixed_income }
oiko_gdp = { value = gdp }
oiko_debt_permille = {
	value = scaled_debt
	multiply = 1000
}
oiko_military_permille = {
	value = military_expenses_share
	multiply = 1000
}

# Debt bar: percent of the credit limit, capped at the bar's 100.
oiko_debt_percent = {
	value = scaled_debt
	multiply = 100
	max = 100
}

# Market goods scope: price over base in per mille (+300 = 30% dearer, -250 = 25% cheaper).
oiko_mg_price_delta_permille = {
	value = market_goods_pricier
	subtract = market_goods_cheaper
	multiply = 1000
}
