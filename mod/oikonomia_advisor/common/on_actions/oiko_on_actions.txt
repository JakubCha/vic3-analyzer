# Oikonomia Advisor: hand the player its journal entries on the monthly pulse.
# Inactive "potential" journal entries only exist in campaigns started with the
# mod, so the entries are added here instead; this also works for older saves.

on_monthly_pulse_country = {
	on_actions = { oiko_monthly_country }
}

oiko_monthly_country = {
	trigger = {
		is_player = yes
	}
	# Budget alarm popup; its own trigger and cooldown decide whether it fires.
	events = {
		oiko.1
	}
	effect = {
		# Dashboard: logs the live line weekly; log one now so the analyzer has data at once.
		if = {
			limit = { NOT = { has_journal_entry = je_oiko_dashboard } }
			add_journal_entry = { type = je_oiko_dashboard }
		}
		oiko_log_status = yes
		# Budget guard: back with every deficit, gone (invalid) once the balance is positive.
		if = {
			limit = {
				net_fixed_income < 0
				NOT = { has_journal_entry = je_oiko_budget_guard }
			}
			add_journal_entry = { type = je_oiko_budget_guard }
		}
		if = {
			limit = {
				c:PRU ?= root
				NOT = { has_journal_entry = je_oiko_pru_opening }
				NOT = { has_variable = oiko_pru_opening_done }
			}
			add_journal_entry = { type = je_oiko_pru_opening }
		}
		if = {
			limit = {
				OR = {
					c:PRU ?= root
					c:NGF ?= root
				}
				NOT = { has_journal_entry = je_oiko_pru_germany }
				NOT = { has_variable = oiko_pru_germany_done }
			}
			add_journal_entry = { type = je_oiko_pru_germany }
		}
		# Idle construction: back whenever the state's queue has under two weeks left.
		if = {
			limit = {
				oiko_construction_levels >= 1
				construction_queue_government_duration < 2
				in_default = no
				NOT = { has_journal_entry = je_oiko_construction_idle }
			}
			add_journal_entry = { type = je_oiko_construction_idle }
		}
		# Cheat menu, only with the game rule on.
		if = {
			limit = {
				has_game_rule = oiko_cheats_on
				NOT = { has_journal_entry = je_oiko_cheats }
			}
			add_journal_entry = { type = je_oiko_cheats }
		}
		# Chapters, one at a time; budget-only help shows none.
		if = {
			limit = { NOT = { has_game_rule = oiko_help_budget } }
			if = {
				limit = {
					game_date < 1856.1.1
					NOT = { has_variable = oiko_ch1_done }
					NOT = { has_journal_entry = je_oiko_chapter_1 }
				}
				add_journal_entry = { type = je_oiko_chapter_1 }
			}
			if = {
				limit = {
					OR = {
						has_variable = oiko_ch1_done
						game_date >= 1850.1.1
					}
					game_date < 1891.1.1
					NOT = { has_journal_entry = je_oiko_chapter_1 }
					NOT = { has_variable = oiko_ch2_done }
					NOT = { has_journal_entry = je_oiko_chapter_2 }
				}
				add_journal_entry = { type = je_oiko_chapter_2 }
			}
			if = {
				limit = {
					OR = {
						has_variable = oiko_ch2_done
						game_date >= 1885.1.1
					}
					NOT = { has_journal_entry = je_oiko_chapter_2 }
					NOT = { has_variable = oiko_ch3_done }
					NOT = { has_journal_entry = je_oiko_chapter_3 }
				}
				add_journal_entry = { type = je_oiko_chapter_3 }
			}
		}
	}
}

# --- Warnings at the moment of a decision ----------------------------------

on_diplomatic_play_started = {
	on_actions = { oiko_on_diplomatic_play_started }
}

# root = the play; the event's own trigger checks the budget.
oiko_on_diplomatic_play_started = {
	effect = {
		scope:initiator ?= {
			if = {
				limit = { is_player = yes }
				trigger_event = { id = oiko.2 }
			}
		}
		scope:target ?= {
			if = {
				limit = { is_player = yes }
				trigger_event = { id = oiko.2 }
			}
		}
	}
}
