# Oikonomia Advisor cheat menu buttons, shown in je_oiko_cheats (game rule oiko_cheats).

oiko_cheat_cash_button = {
	name = "oiko_cheat_cash_button"
	desc = "oiko_cheat_cash_button_desc"
	ai_chance = { value = 0 }
	visible = { has_journal_entry = je_oiko_cheats }
	possible = { always = yes }
	effect = {
		add_treasury = { value = oiko_cheat_cash }
	}
}

oiko_cheat_debt_button = {
	name = "oiko_cheat_debt_button"
	desc = "oiko_cheat_debt_button_desc"
	ai_chance = { value = 0 }
	visible = { has_journal_entry = je_oiko_cheats }
	possible = { scaled_debt > 0 }
	effect = {
		# Money received pays the debt first.
		add_treasury = { value = oiko_cheat_debt }
	}
}
